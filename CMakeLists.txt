cmake_minimum_required(VERSION 3.20)
project(XAO LANGUAGES C CXX)
option(XAO_BUILD_TESTS "Build XAO core tests" ON)
option(XAO_BUILD_AE_PLUGIN "Build native AE plug-in" ON)
set(XAO_AE_SDK_ROOT "" CACHE PATH "Adobe After Effects SDK root")
add_library(xao_core STATIC core/src/Image.cpp core/src/Compositor.cpp core/src/Pipeline.cpp core/src/AsciiEngine.cpp)
target_include_directories(xao_core PUBLIC core/include)
target_compile_features(xao_core PUBLIC cxx_std_17)
set_target_properties(xao_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(xao_core PRIVATE /W4 /permissive- /EHsc)
else()
 target_compile_options(xao_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(XAO_BUILD_TESTS)
 enable_testing()
 add_executable(xao_core_tests tests/core_tests.cpp)
 target_link_libraries(xao_core_tests PRIVATE xao_core)
 add_test(NAME xao_core_tests COMMAND xao_core_tests)
 add_executable(xao_ascii_engine_tests tests/ascii_engine_tests.cpp)
 target_link_libraries(xao_ascii_engine_tests PRIVATE xao_core)
 add_test(NAME xao_ascii_engine_tests COMMAND xao_ascii_engine_tests)
endif()
if(XAO_BUILD_AE_PLUGIN)
 if(NOT WIN32)
  message(FATAL_ERROR "Native .aex build requires Windows x64/MSVC.")
 endif()
 if(NOT XAO_AE_SDK_ROOT OR NOT EXISTS "${XAO_AE_SDK_ROOT}/Headers/AE_Effect.h")
  message(FATAL_ERROR "XAO_AE_SDK_ROOT must point to Adobe SDK root.")
 endif()
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_library(AESDK INTERFACE)
 target_include_directories(AESDK INTERFACE "${XAO_AE_SDK_ROOT}/Headers" "${XAO_AE_SDK_ROOT}/Headers/SP" "${XAO_AE_SDK_ROOT}/Util")
 set(PIPL_RC "${CMAKE_CURRENT_BINARY_DIR}/XAOAsciiPiPL.rc")
 set(PIPL_BIN "${CMAKE_CURRENT_BINARY_DIR}/XAOAsciiPiPL.bin")
 add_custom_command(OUTPUT "${PIPL_RC}" "${PIPL_BIN}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/ae/generate_pipl.py" "${PIPL_RC}" "${PIPL_BIN}"
  DEPENDS ae/generate_pipl.py VERBATIM COMMENT "Generating XAO PiPL resource")
 add_library(XAO_ASCII SHARED ae/XAOAscii.cpp "${PIPL_RC}")
 target_link_libraries(XAO_ASCII PRIVATE xao_core AESDK)
 target_compile_features(XAO_ASCII PRIVATE cxx_std_17)
 target_compile_definitions(XAO_ASCII PRIVATE MSWindows _WINDOWS WIN32 _WIN32 _WIN64 AE_OS_WIN _CRT_SECURE_NO_WARNINGS)
 target_compile_options(XAO_ASCII PRIVATE /W3 /EHsc /MT /O2)
 target_link_options(XAO_ASCII PRIVATE /SUBSYSTEM:WINDOWS,6.00 /MANIFEST:NO "/DEF:${CMAKE_CURRENT_SOURCE_DIR}/ae/XAO.def")
 set_target_properties(XAO_ASCII PROPERTIES PREFIX "" OUTPUT_NAME "XAO_ASCII" SUFFIX ".aex" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin/Release" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin/Release")
endif()
